# onirun's service table (init-lite). Installed as /oni/init.txt in the rootfs.
#   service NAME /path/program args...   socket NAME   oneshot   interface IFACE NAME   start NAME   env K=V
service servicemanager /system/bin/servicemanager
service hwservicemanager /system/bin/hwservicemanager
service logd /system/bin/logd
service keystore2 /system/bin/keystore2 /data/misc/keystore
service apexd /system/bin/apexd
  oneshot
interface aidl/apexservice apexd
service installd /system/bin/installd
service zygote /system/bin/app_process64 -Xzygote /system/bin --zygote --start-system-server --socket-name=zygote
  socket zygote
  socket usap_pool_primary

# aconfig flag storage (/metadata/aconfig): the framework reads flags from the files these write;
# the socket services answer one round of requests and exit (oneshot in aconfigd.rc too)
service aconfigd-init /system/bin/aconfigd-system platform-init
  oneshot
service aconfigd-mainline-init /apex/com.android.configinfrastructure/bin/aconfigd-mainline init
  oneshot
service aconfigd /system/bin/aconfigd-system start-socket
  socket aconfigd_system
  oneshot
service aconfigd-mainline /apex/com.android.configinfrastructure/bin/aconfigd-mainline start-socket
  socket aconfigd_mainline
  oneshot
# stand-ins for daemons the host can't run (userland/stubs/stubsvc.c)
service stubs /system/bin/onistub netd=android.net.INetd dnsresolver=android.net.IDnsResolver+android.net.resolv.unstable.IDnsResolverUnstable@17:b dnsresolver_unstable=android.net.resolv.unstable.IDnsResolverUnstable vold=android.os.IVold
service idmap2d /system/bin/idmap2d
service lmkd /system/bin/lmkd
  socket lmkd
service tombstoned /system/bin/tombstoned
  socket tombstoned_crash
  socket tombstoned_intercept
  socket tombstoned_java_trace

start servicemanager
start hwservicemanager
start aconfigd-init
start aconfigd-mainline-init
start aconfigd
start aconfigd-mainline
start stubs
start lmkd
start tombstoned
start keystore2
start installd
start zygote

# dexopt: artd is started on demand; the job compiles the apps once (userland/data/dexopt.sh)
service artd /apex/com.android.art/bin/artd
  oneshot
interface aidl/artd artd
service dexopt /system/bin/sh /oni/dexopt.sh
  oneshot
start dexopt

# audio: the emulator's HIDL HAL (falls back to null sinks/sources without /dev/snd) and audioserver

service audioserver /system/bin/audioserver
start audio-hal
start audioserver

  oneshot
